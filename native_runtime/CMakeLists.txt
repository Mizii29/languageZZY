cmake_minimum_required(VERSION 3.21)
project(zzy_native VERSION 0.0.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "N0 targets Windows x64 only; other platforms are unverified")
endif()
add_library(zzy_native_options INTERFACE)
option(ZZY_NATIVE_SQLITE "Enable the lazy Windows system SQLite host" ${MSVC})
if(ZZY_NATIVE_SQLITE)
  if(NOT MSVC)
    message(FATAL_ERROR "SQLite SDK host currently requires MSVC; use -DZZY_NATIVE_SQLITE=OFF for core-only MinGW")
  endif()
  target_compile_definitions(zzy_native_options INTERFACE ZZY_NATIVE_SQLITE=1)
endif()
target_include_directories(zzy_native_options INTERFACE include)
target_compile_definitions(zzy_native_options INTERFACE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
if(MSVC)
  target_compile_options(zzy_native_options INTERFACE /W4 /WX /utf-8 /EHsc /permissive- /bigobj)
elseif(MINGW)
  message(WARNING "MinGW is an N0 prototype toolchain; MSVC/clang-cl release remains unverified")
  target_compile_options(zzy_native_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(zzy_native_options INTERFACE -static-libgcc -static-libstdc++)
else()
  message(FATAL_ERROR "Unverified compiler")
endif()
add_executable(zzy-native src/main.cpp)
target_link_libraries(zzy-native PRIVATE zzy_native_options)
if(MSVC)
  target_link_options(zzy-native PRIVATE /STACK:8388608)
else()
  target_link_options(zzy-native PRIVATE -Wl,--stack,8388608)
endif()
if(MINGW)
  target_link_options(zzy-native PRIVATE -municode)
endif()
install(TARGETS zzy-native RUNTIME DESTINATION bin)
option(ZZY_NATIVE_WEB "Build the separate loopback-only native Web host" ${MSVC})
if(ZZY_NATIVE_WEB)
  add_executable(zzy-native-web src/web_main.cpp)
  target_include_directories(zzy-native-web SYSTEM PRIVATE third_party/cpp-httplib)
  target_compile_definitions(zzy-native-web PRIVATE CPPHTTPLIB_REQUEST_URI_MAX_LENGTH=32768 CPPHTTPLIB_HEADER_MAX_LENGTH=32768)
  target_link_libraries(zzy-native-web PRIVATE zzy_native_options ws2_32)
  if(MSVC)
    target_compile_options(zzy-native-web PRIVATE /bigobj)
    target_link_options(zzy-native-web PRIVATE /STACK:8388608)
  else()
    target_link_options(zzy-native-web PRIVATE -municode -Wl,--stack,8388608)
  endif()
  install(TARGETS zzy-native-web RUNTIME DESTINATION bin)
  install(FILES third_party/cpp-httplib/LICENSE DESTINATION share/zzy/licenses RENAME cpp-httplib.txt)
endif()
option(ZZY_NATIVE_GUI "Build the separate native Win32 Sketch GUI" ${MSVC})
option(ZZY_NATIVE_GUI_DESKTOP_TESTS "Run Sketch HWND and pixel tests (interactive desktop required)" OFF)
if(ZZY_NATIVE_GUI)
  add_executable(zzy-native-gui src/main.cpp)
  target_link_libraries(zzy-native-gui PRIVATE zzy_native_options user32 gdi32 ole32 uuid)
  target_compile_definitions(zzy-native-gui PRIVATE ZZY_NATIVE_GUI=1)
  if(MSVC)
    target_link_options(zzy-native-gui PRIVATE /STACK:8388608)
  else()
    target_link_options(zzy-native-gui PRIVATE -municode -Wl,--stack,8388608)
  endif()
  install(TARGETS zzy-native-gui RUNTIME DESTINATION bin)
endif()
if(BUILD_TESTING)
  add_executable(native-runtime-values-tests tests/native/runtime_values.cpp)
  target_link_libraries(native-runtime-values-tests PRIVATE zzy_native_options)
  foreach(case integers floating cycles budgets ownership opaque)
    add_test(NAME runtime-values-${case} COMMAND native-runtime-values-tests "${case}")
    set_tests_properties(runtime-values-${case} PROPERTIES TIMEOUT 60)
  endforeach()
  add_test(NAME runtime-values-empty-selection COMMAND "${CMAKE_COMMAND}" -DEXE=$<TARGET_FILE:native-runtime-values-tests>
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime-selection.cmake")
  add_executable(native-execution-tests tests/native/execution.cpp)
  target_link_libraries(native-execution-tests PRIVATE zzy_native_options)
  foreach(case recovery lazy cycles input resources)
    add_test(NAME execution-${case} COMMAND native-execution-tests "${case}")
    set_tests_properties(execution-${case} PROPERTIES TIMEOUT 30)
  endforeach()
  add_executable(native-library-tests tests/native/library.cpp)
  target_link_libraries(native-library-tests PRIVATE zzy_native_options)
  foreach(case callbacks identity budgets cancellation text files file-commit json sketch-geometry sketch-replay)
    add_test(NAME library-${case} COMMAND native-library-tests "${case}")
    set_tests_properties(library-${case} PROPERTIES TIMEOUT 30)
  endforeach()
  add_test(NAME library-empty-selection COMMAND "${CMAKE_COMMAND}" -DEXE=$<TARGET_FILE:native-library-tests>
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime-selection.cmake")
  find_program(NATIVE_POWERSHELL NAMES pwsh powershell REQUIRED)
  find_program(ZZY_NATIVE_WEB_TEST_PYTHON NAMES python.exe python
    HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../.venv/Scripts" REQUIRED)
  add_test(NAME sketch-geometry-fixed COMMAND "${NATIVE_POWERSHELL}" -NoProfile -File
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-sketch.ps1" -Executable $<TARGET_FILE:zzy-native>)
  set_tests_properties(sketch-geometry-fixed PROPERTIES TIMEOUT 180)
  add_test(NAME sketch-project-fixed COMMAND "${ZZY_NATIVE_WEB_TEST_PYTHON}" -X utf8
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-sketch-project.py" --exe $<TARGET_FILE:zzy-native>
    --python "${ZZY_NATIVE_WEB_TEST_PYTHON}")
  set_tests_properties(sketch-project-fixed PROPERTIES TIMEOUT 60)
  if(ZZY_NATIVE_GUI AND ZZY_NATIVE_GUI_DESKTOP_TESTS)
    add_test(NAME sketch-gui-native COMMAND "${ZZY_NATIVE_WEB_TEST_PYTHON}" -X utf8
      "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-sketch-gui.py" --exe $<TARGET_FILE:zzy-native-gui>
      --core $<TARGET_FILE:zzy-native>)
    set_tests_properties(sketch-gui-native PROPERTIES TIMEOUT 60)
  endif()
  if(ZZY_NATIVE_WEB)
    add_test(NAME web-http COMMAND "${ZZY_NATIVE_WEB_TEST_PYTHON}" -X utf8
      "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-web.py" --exe $<TARGET_FILE:zzy-native-web>
      --sqlite $<IF:$<BOOL:${ZZY_NATIVE_SQLITE}>,ON,OFF>)
    set_tests_properties(web-http PROPERTIES TIMEOUT 120)
  endif()
  if(ZZY_NATIVE_SQLITE)
    add_executable(native-sqlite-tests tests/native/sqlite.cpp)
    if(MSVC)
      target_compile_options(native-sqlite-tests PRIVATE /bigobj)
    endif()
    target_link_libraries(native-sqlite-tests PRIVATE zzy_native_options)
    foreach(case lifecycle cancellation concurrency resources)
      add_test(NAME sqlite-${case} COMMAND native-sqlite-tests "${case}")
      set_tests_properties(sqlite-${case} PROPERTIES TIMEOUT 30)
    endforeach()
    add_test(NAME sqlite-empty-selection COMMAND "${CMAKE_COMMAND}" -DEXE=$<TARGET_FILE:native-sqlite-tests>
      -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime-selection.cmake")
    add_test(NAME sqlite-fixed COMMAND "${NATIVE_POWERSHELL}" -NoProfile -File
      "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-sqlite.ps1" -Executable $<TARGET_FILE:zzy-native>)
    set_tests_properties(sqlite-fixed PROPERTIES TIMEOUT 180)
  else()
    add_test(NAME sqlite-disabled COMMAND "${NATIVE_POWERSHELL}" -NoProfile -File
      "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-sqlite-disabled.ps1" -Executable $<TARGET_FILE:zzy-native>)
  endif()
  add_test(NAME library-fixed COMMAND "${NATIVE_POWERSHELL}" -NoProfile -File
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-library.ps1" -Executable $<TARGET_FILE:zzy-native>)
  set_tests_properties(library-fixed PROPERTIES TIMEOUT 900)
  add_test(NAME file-json-fixed COMMAND "${NATIVE_POWERSHELL}" -NoProfile -File
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-library.ps1" -Executable $<TARGET_FILE:zzy-native> -FileJson)
  set_tests_properties(file-json-fixed PROPERTIES TIMEOUT 900)
  add_test(NAME file-json-integration COMMAND "${NATIVE_POWERSHELL}" -NoProfile -File
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-files.ps1" -Executable $<TARGET_FILE:zzy-native>)
  set_tests_properties(file-json-integration PROPERTIES TIMEOUT 180)
  add_test(NAME runtime-fixed COMMAND "${NATIVE_POWERSHELL}" -NoProfile -File
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-runtime.ps1" -Executable $<TARGET_FILE:zzy-native>)
  set_tests_properties(runtime-fixed PROPERTIES TIMEOUT 900)
  add_test(NAME frontend-fixed COMMAND "${NATIVE_POWERSHELL}" -NoProfile -File
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-fixed.ps1" -Executable $<TARGET_FILE:zzy-native>)
  set_tests_properties(frontend-fixed PROPERTIES TIMEOUT 180)
  add_test(NAME program-fixed COMMAND "${NATIVE_POWERSHELL}" -NoProfile -File
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-check.ps1" -Executable $<TARGET_FILE:zzy-native>)
  set_tests_properties(program-fixed PROPERTIES TIMEOUT 900)
  add_executable(native-program-tests tests/native/program.cpp)
  target_link_libraries(native-program-tests PRIVATE zzy_native_options)
  if(MSVC)
    target_link_options(native-program-tests PRIVATE /STACK:8388608)
  else()
    target_link_options(native-program-tests PRIVATE -Wl,--stack,8388608)
  endif()
  foreach(case ast captures scopes purity literals budgets graph readonly metadata cancellation)
    add_test(NAME program-${case} COMMAND native-program-tests "${case}")
    set_tests_properties(program-${case} PROPERTIES TIMEOUT 30)
  endforeach()
  add_test(NAME program-empty-selection COMMAND "${CMAKE_COMMAND}" -DEXE=$<TARGET_FILE:native-program-tests>
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/selection.cmake")
  add_executable(native-frontend-tests tests/native/frontend.cpp)
  target_link_libraries(native-frontend-tests PRIVATE zzy_native_options)
  foreach(case sha256 utf8 positions source-loader policy diagnostics lexer-boundaries default-boundaries source-defaults cancel-during-scan output cancellation-points)
    add_test(NAME frontend-${case} COMMAND native-frontend-tests "${case}")
    set_tests_properties(frontend-${case} PROPERTIES TIMEOUT 60)
  endforeach()
  # Full 4 Mi-codepoint L-1/L/L+1 scans retain MSVC /RTC1 in Debug.
  # The checked build needs over 60 s; assertions and resource limits stay fixed.
  set_tests_properties(frontend-default-boundaries PROPERTIES TIMEOUT 180)
  add_executable(native-protocol-tests tests/native/protocol.cpp)
  target_link_libraries(native-protocol-tests PRIVATE zzy_native_options)
  add_test(NAME protocol COMMAND native-protocol-tests)
  add_test(NAME cli-info COMMAND "${CMAKE_COMMAND}" -DEXE=$<TARGET_FILE:zzy-native>
    -DMODE=info -DSQLITE=${ZZY_NATIVE_SQLITE} -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake")
  add_test(NAME cli-unsupported COMMAND "${CMAKE_COMMAND}" -DEXE=$<TARGET_FILE:zzy-native>
    -DMODE=test -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake")
  set_tests_properties(protocol cli-info cli-unsupported PROPERTIES TIMEOUT 10)
  add_executable(native-heap-tests tests/native/heap.cpp)
  target_link_libraries(native-heap-tests PRIVATE zzy_native_options)
  foreach(case array-cycle map-cycle closure-cycle reachable-alias quota-gc-retry atomic-limits
      allocation-rollback ownership cancellation unwind-roots iterative-graph repeated-collection)
    add_test(NAME heap-${case} COMMAND native-heap-tests "${case}")
    set_tests_properties(heap-${case} PROPERTIES TIMEOUT 10)
  endforeach()
  add_executable(native-lifecycle-probe tests/native/lifecycle.cpp)
  target_link_libraries(native-lifecycle-probe PRIVATE zzy_native_options)
  foreach(mode success resource cancel error)
    if(mode STREQUAL "success")
      set(expected 0)
    elseif(mode STREQUAL "cancel")
      set(expected 130)
    else()
      set(expected 1)
    endif()
    add_test(NAME lifecycle-${mode} COMMAND "${CMAKE_COMMAND}" -DEXE=$<TARGET_FILE:native-lifecycle-probe>
      -DMODE=${mode} -DEXPECTED=${expected} -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/probe.cmake")
    set_tests_properties(lifecycle-${mode} PROPERTIES TIMEOUT 10)
  endforeach()
endif()
